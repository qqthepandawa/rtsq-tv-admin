String formatCountdown(Duration duration) {
  final total = duration.isNegative ? Duration.zero : duration;
  final hours = total.inHours;
  final minutes = total.inMinutes.remainder(60);
  final seconds = total.inSeconds.remainder(60);
  String two(int value) => value.toString().padLeft(2, '0');
  return hours > 0
      ? '$hours:${two(minutes)}:${two(seconds)}'
      : '${two(minutes)}:${two(seconds)}';
}

/// Indeks sholat terakhir yang sudah masuk (hanya sholat wajib), atau null
/// bila hari ini belum masuk Subuh.
int? activePrayerIndex(DateTime now) {
  int? active;
  for (var i = 0; i < mockPrayers.length; i++) {
    final entry = mockPrayers[i];
    if (!entry.isPrayer) continue;
    if (!entry.at(now).isAfter(now)) active = i;
  }
  return active;
}

/// Indeks sholat berikutnya. Bila sudah lewat Isya, kembali ke Subuh besok.
int nextPrayerIndex(DateTime now) {
  for (var i = 0; i < mockPrayers.length; i++) {
    final entry = mockPrayers[i];
    if (!entry.isPrayer) continue;
    if (entry.at(now).isAfter(now)) return i;
  }
  return mockPrayers.indexWhere((entry) => entry.label == 'SUBUH');
}

DateTime nextPrayerTime(DateTime now) {
  final index = nextPrayerIndex(now);
  final entry = mockPrayers[index];
  final at = entry.at(now);
  if (at.isAfter(now)) return at;
  return entry.at(now.add(const Duration(days: 1)));
}

class _ArchPainter extends CustomPainter {
  const _ArchPainter({required this.color});

  final Color color;

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = color
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2.2;

    for (final offset in [0.0, 46.0]) {
      final w = size.width;
      final h = size.height;
      final left = w * 0.10 + offset;
      final right = w * 0.92 + offset;
      final path = Path()
        ..moveTo(left, h)
        ..lineTo(left, h * 0.52)
        ..quadraticBezierTo(left, h * 0.20, (left + right) / 2, h * 0.06)
        ..quadraticBezierTo(right, h * 0.20, right, h * 0.52)
        ..lineTo(right, h);
      canvas.drawPath(path, paint);
    }
  }

  @override
  bool shouldRepaint(covariant _ArchPainter oldDelegate) =>
      oldDelegate.color != color;
}

import 'package:flutter/material.dart';

class TickerBar extends StatefulWidget {
  const TickerBar({super.key, required this.text, this.startProgress});

  final String text;

  /// Posisi awal marquee (0..1). Dipakai golden test agar tampilan
  /// deterministik; null berarti mulai dari awal.
  final double? startProgress;

  @override
  State<TickerBar> createState() => _TickerBarState();
}

class _TickerBarState extends State<TickerBar>
    with SingleTickerProviderStateMixin {
  late final AnimationController _controller = AnimationController(
    vsync: this,
    duration: const Duration(seconds: 45),
  );

  @override
  void initState() {
    super.initState();
    if (widget.startProgress != null) {
      _controller.value = widget.startProgress!.clamp(0.0, 1.0);
      return;
    }
    _controller.repeat();
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return SizedBox(
      height: 86,
      child: DecoratedBox(
        decoration: const BoxDecoration(
          color: Color(0xFF0E1C18),
          border: Border(
            top: BorderSide(color: AppColors.goldLine),
            bottom: BorderSide(color: AppColors.line),
          ),
        ),
        child: Row(
          children: [
            Container(
              width: 118,
              height: double.infinity,
              color: AppColors.gold,
              alignment: Alignment.center,
              child: const Text(
                'INFO',
                style: TextStyle(
                  fontFamily: AppText.family,
                  fontSize: 22,
                  fontWeight: FontWeight.w800,
                  letterSpacing: 3,
                  color: Color(0xFF0B1714),
                ),
              ),
            ),
            Expanded(
              child: ClipRect(
                child: LayoutBuilder(
                  builder: (context, constraints) {
                    final textWidth = _measure(widget.text);
                    final duration = Duration(
                      milliseconds:
                          ((constraints.maxWidth + textWidth) / 72 * 1000)
                              .round(),
                    );
                    if (_controller.duration != duration) {
                      _controller.duration = duration;
                      if (widget.startProgress == null &&
                          !_controller.isAnimating) {
                        _controller.repeat();
                      }
                    }
                    return AnimatedBuilder(
                      animation: _controller,
                      builder: (context, child) {
                        final x =
                            constraints.maxWidth -
                            _controller.value *
                                (constraints.maxWidth + textWidth);
                        return Transform.translate(
                          offset: Offset(x, 0),
                          child: child,
                        );
                      },
                      child: Text(
                        widget.text,
                        maxLines: 1,
                        softWrap: false,
                        style: AppText.ticker,
                      ),
                    );
                  },
                ),
              ),
            ),
            Container(
              margin: const EdgeInsets.symmetric(horizontal: 18, vertical: 12),
              width: 1,
              color: AppColors.line,
            ),
            Container(
              margin: const EdgeInsets.only(right: 18, top: 6, bottom: 6),
              padding: const EdgeInsets.all(4),
              decoration: BoxDecoration(
                color: const Color(0xFFF7F3E8),
                borderRadius: BorderRadius.circular(10),
              ),
              child: Image.asset(
                'assets/brand/ykca_trim.png',
                height: 62,
                filterQuality: FilterQuality.high,
              ),
            ),
          ],
        ),
      ),
    );
  }

  double _measure(String text) {
    final painter = TextPainter(
      text: TextSpan(text: text, style: AppText.ticker),
      textDirection: TextDirection.ltr,
      maxLines: 1,
    )..layout();
    return painter.width;
  }
}

  p_name text default null,
  p_model text default null,
  p_android_version text default null,
  p_app_version text default null

  device_id uuid,
  status public.device_status,
  pairing_code text,
  expires_at timestamptz

as $$

  v_uid uuid := auth.uid();
  v_id uuid;
  v_status public.device_status;
  v_code text;

    update public.devices d
       set last_seen_at = now(),
           model = coalesce(p_model, d.model),
           android_version = coalesce(p_android_version, d.android_version),
           app_version = coalesce(p_app_version, d.app_version)
     where d.id = v_id;

    return query
      select d.id, d.status, d.pairing_code, d.pairing_expires_at
        from public.devices d
       where d.id = v_id;
    return;
  end if;

  if v_id is not null and v_status = 'revoked' then
    raise exception 'Perangkat ini sudah dicabut aksesnya oleh admin.'
      using errcode = '42501';
  end if;

  -- Buat kode 6 digit yang belum dipakai perangkat pending lain.
  loop
    v_code := lpad((floor(random() * 1000000))::integer::text, 6, '0');
    exit when not exists (
      select 1
        from public.devices d
       where d.pairing_code = v_code
         and d.status = 'pending'
    );
    v_attempt := v_attempt + 1;
    if v_attempt > 50 then
      raise exception 'Gagal membuat kode pairing. Coba lagi.';
    end if;
  end loop;

  if v_id is null then
    insert into public.devices (
      auth_uid, name, status, pairing_code, pairing_expires_at,
      model, android_version, app_version, last_seen_at
    )
    values (
      v_uid,
      coalesce(nullif(p_name, ''), 'TV RTSQ'),
      'pending',
      v_code,
      v_exp,
      p_model,
      p_android_version,
      p_app_version,
      now()
    )
    returning id into v_id;
  else
    update public.devices d
       set name = coalesce(nullif(p_name, ''), d.name),
           status = 'pending',
           pairing_code = v_code,
           pairing_expires_at = v_exp,
           pairing_attempts = 0,
           model = coalesce(p_model, d.model),
           android_version = coalesce(p_android_version, d.android_version),
           app_version = coalesce(p_app_version, d.app_version),
           last_seen_at = now()
     where d.id = v_id;
  end if;

  return query
    select d.id, d.status, d.pairing_code, d.pairing_expires_at
      from public.devices d
     where d.id = v_id;
end;
$$;

-- ---------------------------------------------------------------------------
-- pair_device: admin memasukkan kode yang tampil di TV.
-- ---------------------------------------------------------------------------
create or replace function public.pair_device(
  p_code text,
  p_name text default null
)
returns table (
  device_id uuid,
  device_name text,
  paired_at timestamptz
)
language plpgsql
security definer
set search_path = public
as $$
declare
  v_device public.devices%rowtype;
begin
  if not public.is_admin() then
    raise exception 'Hanya admin yang dapat memasangkan perangkat.'
      using errcode = '42501';
  end if;

  select *
    into v_device
    from public.devices d
   where d.pairing_code = trim(p_code)
     and d.status = 'pending'
   for update;

  if not found then
    raise exception 'Kode pairing tidak dikenal atau sudah dipakai.';
  end if;

  if v_device.pairing_expires_at is null
     or v_device.pairing_expires_at < now() then
    raise exception 'Kode pairing sudah kedaluwarsa. Minta TV menampilkan kode baru.';
  end if;

  if v_device.pairing_attempts >= 10 then
    raise exception 'Terlalu banyak percobaan. Minta TV menampilkan kode baru.';
  end if;

  update public.devices d
     set status = 'paired',
         name = coalesce(nullif(p_name, ''), d.name),
         pairing_code = null,
         pairing_expires_at = null,
         paired_at = now(),
         paired_by = auth.uid(),
         last_seen_at = now()
   where d.id = v_device.id;

  insert into public.audit_log (actor, action, entity, entity_id, diff)
  values (
    auth.uid(),
    'pair',
    'devices',
    v_device.id::text,
    jsonb_build_object('name', coalesce(nullif(p_name, ''), v_device.name))
  );

  return query
    select d.id, d.name, d.paired_at
      from public.devices d
     where d.id = v_device.id;
end;
$$;

returns public.profiles
language plpgsql
security definer
set search_path = public

declare
  v_uid uuid := auth.uid();
  v_email text;
  v_invite public.admin_invites%rowtype;
  v_profile public.profiles%rowtype;
begin
  if v_uid is null then
    raise exception 'Sesi tidak valid.' using errcode = '28000';
  end if;

  select lower(u.email::text)
    into v_email
    from auth.users u
   where u.id = v_uid;

  if v_email is null then
    raise exception 'Akun ini tidak memiliki email.';
  end if;

  select *
    into v_invite
    from public.admin_invites i
   where i.email = v_email::extensions.citext
   for update;

  if not found then
    raise exception 'Email % belum diundang admin.' , v_email
      using errcode = '42501';
  end if;

  if v_invite.expires_at < now() then
    delete from public.admin_invites i where i.email = v_invite.email;
    raise exception 'Undangan sudah kedaluwarsa. Minta admin mengirim ulang.';
  end if;

  insert into public.profiles (id, email, name, role, created_by)
  values (
    v_uid,
    v_email::extensions.citext,
    coalesce(nullif(p_name, ''), split_part(v_email, '@', 1)),
    v_invite.role,
    v_invite.invited_by
  )
  on conflict (id) do nothing;

  select * into v_profile from public.profiles p where p.id = v_uid;

  delete from public.admin_invites i where i.email = v_invite.email;

  return v_profile;
end;
$$;

create or replace function public.set_admin_active(
  p_uid uuid,
  p_active boolean
)
returns public.profiles
language plpgsql
security definer
set search_path = public
as $$
declare
  v_target public.profiles%rowtype;
  v_other_supers integer;
begin
  if not public.is_super() then
    raise exception 'Hanya super admin yang dapat mengubah status admin.'
      using errcode = '42501';
  end if;

  if p_uid = auth.uid() and not p_active then
    raise exception 'Super admin tidak dapat menonaktifkan dirinya sendiri.';
  end if;

  select * into v_target from public.profiles p where p.id = p_uid;
  if not found then
    raise exception 'Admin tidak ditemukan.';
  end if;

  if not p_active and v_target.role = 'super' and v_target.active then
    select count(*)
      into v_other_supers
      from public.profiles p
     where p.role = 'super'
       and p.active
       and p.id <> p_uid;

    if v_other_supers = 0 then
      raise exception 'Harus ada minimal satu super admin yang aktif.';
    end if;
  end if;

  update public.profiles p
     set active = p_active
   where p.id = p_uid
  returning * into v_target;

  return v_target;
end;
$$;

# Supabase — RTSQ TV

Proyek Supabase terpisah dari RTSQ Mobile. Tidak ada Firestore, tidak ada
`service_role` di aplikasi maupun repo.

## Isi migrasi

Jalankan **berurutan** di SQL Editor (atau `supabase db push`):

| Berkas | Isi |
|---|---|
| `0001_extensions.sql` | `pgcrypto`, `citext` |
| `0002_types.sql` | enum `admin_role`, `device_status`, `content_type`, `content_status`, `command_status`, `log_level` |
| `0003_tables.sql` | tabel, indeks, trigger `updated_at`/`version`, audit |
| `0004_functions.sql` | helper RLS + RPC pairing, heartbeat, commands, invite, admin |
| `0005_rls.sql` | RLS + policy tiap tabel |
| `0006_storage.sql` | bucket privat `tv-media` + policy |
| `0007_realtime.sql` | publikasi `contents`, `ticker`, `settings`, `commands` |
| `0008_seed.sql` | baris `settings` Batam + ticker sambutan |

## Setelah migrasi

## Yang dibutuhkan aplikasi

Hanya dua nilai, keduanya aman dipublikasikan:

- `SUPABASE_URL` — `https://<project-ref>.supabase.co`
- `SUPABASE_ANON_KEY` — anon public key

**Jangan pernah** memakai/menyimpan `service_role`.

## Alur perangkat (TV)

1. TV memanggil `register_device()` → dapat `pairing_code` 6 digit, berlaku 10 menit.
2. Admin membuka panel → masukkan kode → RPC `pair_device(code)`.
3. TV polling `device_heartbeat()`; bila `status = paired`, TV mengunduh snapshot.
4. Perintah dari admin diambil lewat `claim_commands()`, ditutup dengan
   `complete_command()`.

## Catatan keamanan

- Semua tabel memakai RLS. Perangkat paired hanya bisa **membaca** konten
  `published` + `active`, ticker aktif, dan settings.
- Perangkat tidak punya policy `UPDATE` bebas; perubahan status hanya lewat RPC
  `security definer`.
- `audit_log` hanya bisa dibaca admin; penulisan lewat trigger definer.
- Storage `tv-media` privat; TV memakai signed URL 7 hari dan menyimpan cache.
